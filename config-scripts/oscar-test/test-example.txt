def runOscarTest(String testId, String testPath, boolean installCli = false, boolean installPython = false) {
    node('docker') {
        docker.image('ghcr.io/grycap/oscar-tests:min').inside('--shm-size=1g -u root') {
            env.BROWSER = 'chromium'
            env.ROBOT_TESTS_DIR = "${WORKSPACE}/${testPath}"
            env.ROBOT_REPORTS_DIR = "${WORKSPACE}/results_"
            env.ROBOT_TEST_RUN_ID = "${testId}"

            unstash 'oscar-tests'

            if (installCli) {
                retry(2) {
                    sh 'go install github.com/grycap/oscar-cli/v2@v<<oscar-cli-version>>'
                }
                sh 'mv /root/go/bin/oscar-cli /bin'
            }

            if (installPython) {
                retry(2) {
                    sh 'pip install oscar-python==<<oscar-python-version>>'
                }
            }

            withCredentials([file(credentialsId: 'imagine-secret-file', variable: 'IMAGINE_FILE'),
                             file(credentialsId: 'keycloak-secret-file', variable: 'KEYCLOAK_FILE')]) {
                env.ROBOT_OPTIONS = "--pythonpath . --variablefile $IMAGINE_FILE  --variablefile $KEYCLOAK_FILE"
                retry(2) {
                    if (installPython) {
                        sh """
                            OSCAR_TEST_AUTH_FILE=${KEYCLOAK_FILE} \
                            OSCAR_TEST_CLUSTER_FILE=${IMAGINE_FILE} \
                            python3 -m pytest ${WORKSPACE}/tests/python_tests/ -v --tb=short
                        """
                    }else{
                        sh '/opt/robotframework/bin/run-tests-in-virtual-screen.sh'
                    }
                }
            }
        }
       
    }
}
